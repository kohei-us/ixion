%% Test circular dependency detection through table references.  E1
%% references the data area of Table1 which contains D6, which in turn
%% references E1, forming a cycle.  D9 is not part of the cycle but
%% references the data area containing D6, so the error propagates to it.
%mode init
C3@Category
C4@A
C5@B
C6@C
C7@D
C8@E
C9@Total
D3@Value
D4:1
D5:2
D6=E1+1
D7:4
D8:5
D9=SUBTOTAL(109,[Value])
E1=SUM(Table1[Value])
E2=COUNTA(Table1[#Headers])
%mode table
name=Table1
range=C3:D9
columns=Category,Value
totals-row-count=1
%push
%calc
%mode result
D6=#REF!
D9=#REF!
E1=#REF!
E2=2
%check
%exit
