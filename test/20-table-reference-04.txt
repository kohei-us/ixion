%% Table references get rewritten to reference the cloned table when a
%% sheet is copied.
%mode init
C3@Category
C4@A
C5@B
C6@C
C7@D
C8@E
C9@Total
D3@Value
D4:1
D5:2
D6:3
D7:4
D8:5
D9=SUBTOTAL(109,[Value])
A1=SUM(Table1[Value])
%mode table
name=Table1
range=C3:D9
columns=Category,Value
totals-row-count=1
%push
%calc
%mode result
D9=15
A1=15
%check
%copy-sheet sheet copy
%mode result
copy!D9=15
copy!A1=15
%check
%% Editing a value inside the source table should affect only the formula
%% cells on the source sheet.
%mode edit
D6:100
%recalc
%mode result
A1=112
D9=112
copy!A1=15
copy!D9=15
%check
%% Editing a value inside the cloned table should affect only the formula
%% cells on the copied sheet.
%mode edit
copy!D6:200
%recalc
%mode result
A1=112
D9=112
copy!A1=212
copy!D9=212
%check
%exit
