%% Copying a sheet with a multi-column array formula must register the
%% copied group through its top-left cell only.
%mode session
insert-sheet:src
%mode init
A1=1
B1=2
A2=3
B2=4
{D1:E2}{=A1:B2*10}
%calc
%mode result
D1=10
E1=20
D2=30
E2=40
%check
%copy-sheet src copy
%mode result
copy!D1=10
copy!E1=20
copy!D2=30
copy!E2=40
%check
%% Editing a value on the copy must re-calculate the whole copied group and
%% leave the source group alone.
%mode edit
copy!A1:5
%recalc
%mode result
copy!D1=50
copy!E1=20
copy!D2=30
copy!E2=40
src!D1=10
%check
%exit
